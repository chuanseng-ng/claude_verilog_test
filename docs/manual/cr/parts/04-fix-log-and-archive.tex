\subsubsection{Issues Fixed}\label{cr:fix:fixes-index:issues-fixed}

\paragraph{Bug 1: Incorrect AXI Read Timing in FETCH State}\label{cr:fix:fixes-index:bug-1-incorrect-axi-read-timing-in-fetch-state}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem}: Code assumed \code{rvalid} asserts same cycle as
  \code{arready} (violates AXI protocol)
\item
  \textbf{Impact}: CPU hung waiting for impossible condition
\item
  \textbf{Fix}: Changed to wait only for \texttt{rvalid\ \&\&\ rready}
  (data phase), independent of address phase
\item
  \textbf{Files}: \code{rv32i_control.sv} (FETCH state)
\end{itemize}

\end{record}

\paragraph{\texorpdfstring{Bug 2: Incorrect AXI Read/\allowbreak{}Write
Timing in MEM\_WAIT
State}{Bug 2: Incorrect AXI Read/Write Timing in MEM\_WAIT State}}\label{cr:fix:fixes-index:bug-2-incorrect-axi-readwrite-timing-in-mem-wait-state}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem}: Same protocol violation for load/\allowbreak{}store
  operations
\item
  \textbf{Impact}: CPU hung on all memory operations
\item
  \textbf{Fix}: Wait only for data phase completion
  (\code{rvalid}/\code{bvalid})
\item
  \textbf{Files}: \code{rv32i_control.sv} (MEM\_WAIT state)
\end{itemize}

\end{record}

\paragraph{Bug 3: Missing AXI Address Mux}\label{cr:fix:fixes-index:bug-3-missing-axi-address-mux}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem}: \code{axi_araddr} always used PC (not multiplexed
  for data accesses)
\item
  \textbf{Impact}: All loads fetched from wrong address (PC instead of
  calculated address)
\item
  \textbf{Fix}: Added \code{data_access} control signal and address mux
\item
  \textbf{Files}: \code{rv32i_control.sv}, \code{rv32i_core.sv}
\end{itemize}

\end{record}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem}: Branch decision computed but not registered before
  state transition
\item
  \textbf{Impact}: JAL backward jumps failed, branch tests unreliable
\item
  \textbf{Fix}: Added \code{branch_decision_reg} to latch decision at
  end of EXECUTE state
\item
  \textbf{Files}: \code{rv32i_control.sv}
\item
  \textbf{Tests Fixed}: JAL backward jump, all 6 branch instructions

\begin{record}

\item
  \textbf{Problem}: Load data not latched from AXI read response
\item
  \textbf{Impact}: All 5 load instructions failed
\item
  \textbf{Fix}: Added \code{mem_rdata_raw} register to capture
  \code{axi_rdata}
\item
  \textbf{Files}: \code{rv32i_core.sv}

  \textbf{Tests Fixed}: LW, LH, LHU, LB, LBU (all 5 load instructions)

\subsection{\texorpdfstring{11. I2C START/\allowbreak{}STOP Condition
Times (bead gecv)}{11. I2C START/STOP Condition Times (bead gecv)}}\label{cr:fix:fixes-index:11-i2c-startstop-condition-times-bead-gecv}

\begin{record}

\textbf{File}: \src{rtl/periph/i2c_bit_engine.sv} (tick reload for
\code{S_ST_LO}, \code{S_RS_HI}, \code{S_STP_HI}, \code{S_STP_FREE})
\textbf{Date}: 2026-10-10 \textbf{Severity}: P3 (no external I2C slave
is attached in any flow; software could mask it with a delay)
\textbf{Found by}: bead pnfw item 3 (verification-orchestrator),
measured on the bus

\end{record}

\subsubsection{Issue Fixed}\label{cr:fix:fixes-index:issue-fixed-4}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem}: every START/\allowbreak{}STOP condition phase lasted
  ONE tick (N = CLKDIV+1 clk, a quarter bit period), about half the I2C
  minimum at the Standard divisor. Measured at 100 MHz clk: tBUF 2.53 us
  at CLKDIV=249 (min 4.7) and 0.66 us at CLKDIV=62 (min 1.3); tSU;STO
  2.53 us (min 4.0, Fast 0.66 vs 0.6); tHD;STA 2.50 us (min 4.0);
  tSU;STA of a repeated START 2.53 us (min 4.7). The Fast-mode
  tHD;STA/\allowbreak{}tSU;STA/\allowbreak{}tSU;STO met their 0.6 us by
  only 0.03..0.06 us.
\item
  \textbf{Impact}: a driver that wrote the next CMD on the first APB
  access after \code{STATUS.busy} dropped produced a
  STOP-\textgreater START gap below spec; a Standard-mode slave could
  miss the START or STOP.
\item
  \textbf{Fix}: the phase lengths are now whole-tick counts set by the
  tick reload (no new state, \code{state_e} still 17 names / 5 bits):
  \code{S_ST_LO}, \code{S_RS_HI}, \code{S_STP_HI} = 2 ticks (reload
  \code{2N-1}), \code{S_STP_FREE} = 3 ticks (\code{3N-1}). 2 ticks = 5.0
  us at 100 kHz clears the tightest Standard minimum (4.7 us); at 400
  kHz 2 ticks = 1.25 us clears 0.6 us but NOT tBUF\textquotesingle s 1.3
  us, hence 3 ticks (1.875 us; Standard 7.5 us). Because a tick is
  exactly 1/(4 f\_scl), the guarantee holds at any CLKDIV for a
  Standard-mode bus \textless= 100 kHz and a Fast-mode bus \textless=
  400 kHz; it does not cover \textgreater{} 400 kHz and tBUF is enforced
  only after this master\textquotesingle s own STOP (derivation table in
  the module header).
\item
  \textbf{Busy semantics}: the bus-free interval is INSIDE
  \code{STATUS.busy} and the DONE event of a STOP command is raised at
  its end (3 ticks after the STOP edge instead of 1). Commands written
  during the interval are ignored like any busy-time command, so tBUF
  holds even for a command written on the first APB access after busy
  drops.
\item
  \textbf{Not changed}: data-bit timing (period exactly 4N,
  tLOW/\allowbreak{}tHIGH) and
  \texttt{K\ =\allowbreak{}\ floor(N/\allowbreak{}8)} (bead \bead{cd15} stays
  deferred).
\item
  \textbf{Cost}: \code{tick_q} 17 -\textgreater{} 18 bits (+1 flop;
  \code{S_STP_FREE} reload is \code{0x2FFFF} at CLKDIV=0xFFFF), one
  18-bit constant-3 adder in the reload mux, no new states.
\item
  \textbf{Tests}: \code{test_i2c_stop_to_start_bus_free_time_meets_spec}
  (was \code{expect_fail=True}, limits unchanged), new
  \code{test_i2c_start_stop_condition_times_meet_spec} (tSU;STO, tHD;STA
  on START and repeated START, tSU;STA, strict spec limits, both
  divisors), \code{test_i2c_busy_covers_bus_free_time},
  \code{test_i2c_condition_tick_reloads_at_clkdiv_ffff}. All red on the
  old RTL, green after.
\end{itemize}

\subsubsection{Issue Fixed}\label{cr:fix:fixes-index:issue-fixed-8}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem (traced)}: after a divergence-stack overflow,

  \code{gpu_compute_unit.gpu_error_o} (cleared only by \code{rst_n})

\subsection{16. CSR Write to minstret / minstreth Lands Under In-Flight

\begin{record}

\textbf{Files}: \src{rtl/cpu/core/rv32i_csr_file.sv} (writer scoreboard,
\code{inflight_older_i}), \src{rtl/cpu/core/rv32i_core.sv}
(\code{csr_older_inflight}) \textbf{Date}: 2026-10-11 \textbf{Severity}:
P3 (software-visible only when firmware WRITES
\code{minstret}/\code{minstreth}, e.g. zeroing before a measurement: the
count read high by 1-2) \textbf{Found by}: bead kiit (fix \#14),
deliberately split out; user decision 2026-10-11: fix in RTL so the
written value wins.

\end{record}

\subsubsection{Issue Fixed}\label{cr:fix:fixes-index:issue-fixed-9}

\begin{record}

\begin{itemize}
\tightlist
\item
  \textbf{Problem (pipeline terms)}: a CSR write is applied in EX1a, but
  \code{retire_i} (\code{commit_valid}) fires in WB, five register
  stages later. At the write, up to one older instruction (EX-to-EX
  spacing is \textgreater= 3 cycles) is still in flight and the writer
  itself has not retired; both retirement strobes then incremented on
  top of the written value. RISC-V privileged spec: older instructions
  retire before the write and the writing instruction\textquotesingle s
  own retirement must not increment. Measured on the unfixed RTL:
  \texttt{csrrw\ minstret,\allowbreak{}\ V} followed directly by
  \texttt{csrrs\ rd,\allowbreak{}\ minstret,\allowbreak{}\ x0} read
  \code{V+1} (older ADDI in flight), and
  \texttt{csrrw\ minstret,\allowbreak{}x0\ ;\ csrrw\ minstreth,\allowbreak{}x0}
  left the low word at 1.
\item
  \textbf{Fix}: a two-entry writer scoreboard in \code{rv32i_csr_file}.
  Each accepted write (once, on the cycle it leaves EX:
  \code{!csr_stall_i}; not squashed by trap/\allowbreak{}MRET;
  \code{csr_wr_en}, so
  \texttt{CSRRS/\allowbreak{}CSRRC\ rs1=\allowbreak{}x0} and
  \code{uimm=0} forms are not writes) records the number of retire
  strobes up to and including the writer:
  \texttt{inflight\_\allowbreak{}older\_\allowbreak{}i\ +\allowbreak{}\ 1},
  where \code{inflight_older_i} is the count of valid instructions in
  the four registers after EX1a (the WB-stage instruction retires in the
  write cycle itself, which the write overrides). Strobes are then
  classified in program order: the writer\textquotesingle s own strobe
  never increments (neither word); strobes older than a low-word write
  (\code{minstret}) are dropped; strobes older than a high-word write
  (\code{minstreth}) still count in the low word but their carry into
  the written high word is discarded. Two entries cover the two writers
  that can overlap (\texttt{csrw\ minstret\ ;\ csrw\ minstreth} is the
  common zeroing pair); the entry is freed when its own strobe is seen.
  A trap entry does not touch the scoreboard: the EX-stage trap point is
  two stages past EX and EX-to-EX spacing is \textgreater= 3, so no
  younger writer has executed when a trap is taken and every pending
  writer is older and still retires (clearing on trap entry is
  behaviourally equivalent in this pipeline: mutant M10 survives).
\item
  \textbf{Counter write-enable cone}: the 64-bit \code{minstret} enable
  gains ONE flop input (\code{!mrw_drop_q}) and the upper-half carry one
  (\code{!mrw_nocarry_q}). Both are registered flags precomputed from
  the scoreboard\textquotesingle s next state, so no comparator or
  CSR-decode term sits between \code{retire_i} and the 64 counter flops.
  The new logic (alloc decode, 3-bit position countdown,
  \code{inflight_older_i} 4-input popcount) feeds only 12 new flops.
\item
  \textbf{What the same analysis gives for the other counters} (no
  change needed): \code{mcycle}/\code{mcycleh} count clocks, not
  retirements, so there is no in-flight term; the write beats that
  cycle\textquotesingle s increment (kiit) and the counter then runs
  from the written value (read d instructions later = V + EX-to-EX
  distance - 1, checked in \code{test_mcycle_write_wins_and_counts_on},
  green before and after). \code{mhpmcounter3-5} count physical events
  in the cycle they occur: the write beats an event in its own cycle; an
  older load\textquotesingle s D\$ miss strobe that arrives after the
  write counts on top of it (time-ordered, not program-ordered; the
  privileged spec has no program-order rule for hpm events). Left
  unchanged (user decision 2026-10-11: hpm writes stay time-ordered).
\item
  \textbf{Not changed (accepted, user decision 2026-10-11)}: a CSR read
  does not see the retirement of its immediate predecessor (the retire
  strobe lands 5 cycles after EX, the next EX is 3 cycles later):
  \code{csrr} reads trail the architectural count by at most one. This
  is pinned by \code{test_minstret_matches_retirements_between_reads}
  and allowed for (never more) in the f7z3 spec checks.
\item
  \textbf{Reference model}: \src{tb/models/csr_model.py}
  \code{CsrModel.tick()} models the absorption by program-order index
  (\code{access_seq}); \src{tb/models/rv32i_model.py} carries no CSR
  state at all (CSR reads return 0), so it has no minstret semantics to
  correct. The two existing minstret-writing sweeps in
  \code{test_csr_access} encoded the defect and went red on the unfixed
  RTL once the oracle was corrected.
\item
  \textbf{Tests}: \src{tb/cocotb/cpu/test_minstret_write.py} (11 tests)
  plus the oracle-fed \code{test_csr_access} (15). See the PR for the
  red-\textgreater green evidence and the mutant table.
\item
  \textbf{Macros}: the committed Sky130 and ASAP7 CPU hard-macro views
  predate this fix and keep the old behaviour until the CPU macro is
  re-hardened (bead \bead{j41m.4}).
\end{itemize}

\subsection{Issue Identified}\label{cr:fix:critical-fixes:issue-identified}

\begin{record}

User identified critical bugs in the AXI4-Lite protocol handling in
\code{rv32i_control.sv} and \code{rv32i_core.sv}.

\end{record}

\subsection{Bug 1: Incorrect AXI Read Timing in FETCH State}\label{cr:fix:critical-fixes:bug-1-incorrect-axi-read-timing-in-fetch-state}

\subsubsection{Problem}\label{cr:fix:critical-fixes:problem}

\begin{record}

Original code in \code{rv32i_control.sv} FETCH state:

\begin{lstlisting}[language=Verilog]
FETCH: begin
  // INCORRECT: Assumes rvalid asserts same cycle as arready
  if (axi_arready && axi_arvalid) begin
    if (axi_rvalid) begin
      // ...
    end
  end
end
\end{lstlisting}

\textbf{Why this is wrong}:

In AXI4-Lite protocol, the address phase and data phase are
\textbf{independent}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Address phase}: \texttt{arvalid\ \&\&\ arready} → address
  handshake complete
\item
  \textbf{Data phase}: \texttt{rvalid\ \&\&\ rready} → data handshake
  complete
\end{enumerate}

\textbf{\code{rvalid} can come many cycles after \code{arready}}
depending on memory latency. The original code would only transition to
DECODE if all three signals (\code{arready}, \code{arvalid},
\code{rvalid}) were high in the same cycle, which violates the protocol
and causes the CPU to hang waiting for an impossible condition.

\end{record}

\subsubsection{Fix}\label{cr:fix:critical-fixes:fix}

\begin{record}

Updated FETCH state (rv32i\_\allowbreak{}control.sv:75-88):

\begin{lstlisting}[language=Verilog]
FETCH: begin
  // Wait for AXI read data phase to complete
  // Note: Address phase (arvalid/arready) and data phase (rvalid/rready)
  // are independent. rvalid can come many cycles after arready.
  if (axi_rvalid && axi_rready) begin
    // Data received, check for AXI error
    if (axi_rresp != 2'b00) begin
      next_state = TRAP;  // AXI error -> illegal instruction trap
    end else begin
      next_state = DECODE;
    end
  end
  // Otherwise stay in FETCH state waiting for rvalid

  // Debug halt request during fetch
  if (dbg_halt_req) begin
    next_state = HALTED;
  end
end
\end{lstlisting}

\textbf{Key changes}:

\begin{itemize}
\tightlist
\item
  \ok{} Only check \texttt{rvalid\ \&\&\ rready} for data phase
  completion
\item
  \ok{} Don\textquotesingle t check \code{arready} in the transition
  condition (address phase happens separately via output logic)
\item
  \ok{} Stay in FETCH state until data arrives (can be many cycles
  later)
\end{itemize}

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{Bug 3: Missing AXI Address Mux}\label{cr:fix:critical-fixes:bug-3-missing-axi-address-mux}

\subsubsection{Problem}\label{cr:fix:critical-fixes:problem-2}

\begin{record}

Original code in \code{rv32i_core.sv}:

\begin{lstlisting}[language=Verilog]
// INCORRECT: Always uses PC for read address
assign axi_araddr = pc_reg;
\end{lstlisting}

\textbf{Why this is wrong}:

The CPU uses a unified instruction/\allowbreak{}data bus (AXI4-Lite), so
\code{axi_araddr} must be multiplexed:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{During FETCH}: \code{araddr} = \code{pc_reg} (fetching
  instruction)
\item
  \textbf{During MEM\_WAIT (loads)}: \code{araddr} = \code{mem_addr}
  (loading data from calculated address)
\end{enumerate}

Without this mux, loads would incorrectly use the PC as the data
address, causing all loads to fetch from the wrong memory location.

\end{record}

\subsubsection{Fix}\label{cr:fix:critical-fixes:fix-2}

\begin{record}

Added control signal \code{data_access} to FSM
(rv32i\_\allowbreak{}control.sv:46):

\begin{lstlisting}[language=Verilog]
output logic        data_access      // 1=data access (MEM_WAIT), 0=instruction fetch (FETCH)
\end{lstlisting}

Drive \code{data_access} high in MEM\_WAIT state
(rv32i\_\allowbreak{}control.sv:270):

\begin{lstlisting}[language=Verilog]
MEM_WAIT: begin
  data_access = 1'b1;  // Data access mode (not instruction fetch)
  // ...
end
\end{lstlisting}

Use \code{data_access} to mux \code{axi_araddr}
(rv32i\_\allowbreak{}core.sv:143-146):

\begin{lstlisting}[language=Verilog]
// AXI read address mux
// Select between instruction fetch (PC) and data access (mem_addr)
// - data_access=0 (FETCH state): araddr = PC (fetching instruction)
// - data_access=1 (MEM_WAIT state): araddr = mem_addr (loading data)
assign axi_araddr = data_access ? mem_addr : pc_reg;
\end{lstlisting}

\textbf{Key changes}:

\begin{itemize}
\tightlist
\item
  \ok{} Added \code{data_access} control signal from FSM
\item
  \ok{} Mux \code{axi_araddr} based on CPU state
\item
  \ok{} FETCH uses PC, MEM\_WAIT uses calculated address
\end{itemize}

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{AXI Protocol Summary}\label{cr:fix:critical-fixes:axi-protocol-summary}

\subsubsection{Correct AXI4-Lite Read Transaction}\label{cr:fix:critical-fixes:correct-axi4-lite-read-transaction}

\begin{record}

\begin{lstlisting}[language={}]
Cycle 0:  Master asserts arvalid, araddr
Cycle 0:  Slave may or may not assert arready
Cycle N:  Slave asserts arready (address handshake complete)
          Master must keep arvalid high until arready seen
Cycle M:  Slave asserts rvalid, rdata, rresp (M >= N, can be many cycles later)
          Master asserts rready to accept data
Cycle M:  Data handshake complete (rvalid && rready)
\end{lstlisting}

\textbf{Key insight}: Address phase (cycle N) and data phase (cycle M)
are independent. M can be \textgreater\textgreater{} N.

\end{record}

\subsubsection{Correct AXI4-Lite Write Transaction}\label{cr:fix:critical-fixes:correct-axi4-lite-write-transaction}

\begin{record}

\begin{lstlisting}[language={}]
Cycle 0:  Master asserts awvalid, awaddr, wvalid, wdata, wstrb
Cycle 0:  Slave may or may not assert awready, wready
Cycle N:  Slave asserts awready (address handshake)
Cycle P:  Slave asserts wready (data handshake)
Cycle Q:  Slave asserts bvalid, bresp (write response, Q >= max(N,P))
          Master asserts bready to accept response
Cycle Q:  Response handshake complete (bvalid && bready)
\end{lstlisting}

\subsection{Root Cause}\label{cr:fix:rtl-isa-test-fix:root-cause}

\begin{record}

The ISA compliance tests load the program to memory \textbf{AFTER}
reset, when the CPU has already started executing. This causes:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  CPU resets with PC=0
\item
  CPU enters FETCH state and fetches from PC=0
\item
  Memory is \textbf{empty} (returns 0x00000000)
\item
  CPU decodes 0x00000000 (likely illegal or NOP)
\item
  CPU advances PC to 0x04
\item
  Test halts CPU (PC is now at 0x04)
\item
  Test loads the actual program to memory
\item
  Test attempts to write PC=0 via debug interface
\item
  \textbf{Debug PC write fails} (reason TBD - possible
  timing/\allowbreak{}synchronization issue)
\item
  Test resumes from PC=0x04 instead of PC=0x00
\item
  First instruction executed is at PC=0x04 (the instruction that should
  be SKIPPED)
\item
  Test fails because wrong instruction executed
\end{enumerate}

\end{record}

\section{EBREAK Halt Bug Fix}\label{cr:fix:ebreak-bug-fix}

\begin{record}

\textbf{Date}: 2026-01-28 \textbf{Issue}: EBREAK instruction does not
properly set halt cause in status register \textbf{Status}: FIXED

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{Problem Description}\label{cr:fix:ebreak-bug-fix:problem-description}

\begin{record}

The EBREAK instruction (0x00100073) \textbf{does cause the CPU to halt},
but the halt \textbf{cause code} in the debug status register is
reported incorrectly (shows 0x0 instead of 0x8).

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{Root Cause}\label{cr:fix:ebreak-bug-fix:root-cause}

\subsubsection{\texorpdfstring{The Bug
(rv32i\_\allowbreak{}cpu\_\allowbreak{}top.sv:325-340)}{The Bug (rv32i\_cpu\_top.sv:325-340)}}\label{cr:fix:ebreak-bug-fix:the-bug-rv32i-cpu-topsv325-340}

\begin{record}

The original \code{ebreak_halt} flag detection logic was:

\begin{lstlisting}[language=Verilog]
if (commit_valid && (commit_insn == 32'h00100073) && !dbg_halted) begin
  ebreak_halt <= 1'b1;
end
\end{lstlisting}

\end{record}

\subsubsection{Why This Failed}\label{cr:fix:ebreak-bug-fix:why-this-failed}

\begin{record}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{EBREAK execution path}: EXECUTE → HALTED (skips WRITEBACK)
\item
  \textbf{commit\_\allowbreak{}valid} is only asserted in WRITEBACK
  state (rv32i\_\allowbreak{}control.sv:363)
\item
  \textbf{Result}: EBREAK never commits, so \code{commit_valid} is never
  true for EBREAK
\item
  \textbf{Consequence}: \code{ebreak_halt} flag never gets set
\end{enumerate}

\end{record}

\subsubsection{What Actually Happened}\label{cr:fix:ebreak-bug-fix:what-actually-happened}

\begin{record}

\begin{textblock}
State Machine Behavior:
  FETCH    → Fetches EBREAK (0x00100073)
  DECODE   → Sets ebreak = 1
  EXECUTE  → Sees ebreak=1, transitions to HALTED (skips WRITEBACK!)
  HALTED   → dbg_halted = 1, but ebreak_halt = 0 ❌

Status Register (bits [7:4] = halt cause):
  if (ebreak_halt)        → 0x8 ✓ EBREAK
  else if (bp0_hit)       → 0x2   Breakpoint 0
  else if (bp1_hit)       → 0x3   Breakpoint 1
  else if (dbg_halt_req)  → 0x1   Debug halt request
  else if (dbg_step_req)  → 0x4   Single-step
  else                    → 0x0 ❌ (reported for EBREAK)
\end{textblock}

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{The Fix}\label{cr:fix:ebreak-bug-fix:the-fix}

\subsubsection{New Detection Logic}\label{cr:fix:ebreak-bug-fix:new-detection-logic}

\begin{record}

\begin{lstlisting}[language=Verilog]
logic prev_halted;

always_ff @(posedge clk or negedge rst_n) begin
  if (!rst_n) begin
    ebreak_halt <= 1'b0;
    prev_halted <= 1'b0;
  end else begin
    prev_halted <= dbg_halted;

    // Set when transitioning into HALTED state with EBREAK instruction
    if (!prev_halted && dbg_halted && (commit_insn == 32'h00100073)) begin
      ebreak_halt <= 1'b1;
    // Clear when CPU resumes
    end else if (!dbg_halted) begin
      ebreak_halt <= 1'b0;
    end
  end
end
\end{lstlisting}

\end{record}

\subsubsection{How It Works}\label{cr:fix:ebreak-bug-fix:how-it-works}

\begin{record}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Track halted state}: \code{prev_halted} registers the previous
  cycle\textquotesingle s halt status
\item
  \textbf{Detect transition}:
  \texttt{!prev\_\allowbreak{}halted\ \&\&\ dbg\_\allowbreak{}halted}
  detects entry into HALTED state
\item
  \textbf{Check instruction}:
  \texttt{commit\_\allowbreak{}insn\ =\allowbreak{}=\allowbreak{}\ 0x00100073}
  verifies it\textquotesingle s EBREAK
\item
  \textbf{Set flag}:
  \texttt{ebreak\_\allowbreak{}halt\ <=\allowbreak{}\ 1'b1}
\end{enumerate}

\end{record}

\subsubsection{Why This Works}\label{cr:fix:ebreak-bug-fix:why-this-works}

\begin{record}

\begin{itemize}
\tightlist
\item
  \code{commit_insn} is wired directly to the instruction register
  (rv32i\_\allowbreak{}core.sv:454)
\item
  When CPU halts due to EBREAK, instruction register still contains
  EBREAK
\item
  Instruction register only updates on fetch, not during halt
\item
  Our logic detects the HALTED state entry and checks what instruction
  caused it
\end{itemize}

\end{record}

\begin{record}

During random instruction tests, hundreds of PC mismatch errors were
logged:

\begin{lstlisting}[language={}]
ERROR PC mismatch: RTL=0x00000000, Model=0x0000018c
ERROR PC mismatch: RTL=0x00000004, Model=0x00000190
ERROR PC mismatch: RTL=0x00000008, Model=0x00000194
...
\end{lstlisting}

Despite these errors, tests still reported \textbf{100\% pass rate} with
\textbf{0 mismatches}, creating confusion about the actual test status.

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{Root Cause}\label{cr:fix:pc-mismatch-fix:root-cause}

\subsubsection{The Bug}\label{cr:fix:pc-mismatch-fix:the-bug}

\begin{record}

\textbf{Orphaned monitor tasks accumulating across test seeds}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  Each seed starts a new \code{monitor_commits} coroutine:

  \begin{lstlisting}[language=Python]
  cocotb.start_soon(monitor_commits(dut, scoreboard=scoreboard, count=commit_count))
  \end{lstlisting}
\item
  Monitor tasks run indefinitely (infinite \texttt{while\ True} loop)
\item
  When a seed completes, its monitor task \textbf{continues running}
\item
  Next seed starts with fresh model (PC=0x0000), but old monitors still
  have stale models
\item
  Result: Multiple monitors checking the same commits with different
  model states
\end{enumerate}

\end{record}

\subsubsection{The Accumulation}\label{cr:fix:pc-mismatch-fix:the-accumulation}

\begin{record}

\begin{textblock}
Seed 0: Creates monitor₀ (model PC ends at 0x18c)
  ↓
Seed 1: Creates monitor₁ (model PC starts at 0x0)
        Now 2 monitors running!
  ↓
Seed 2: Creates monitor₂ (model PC starts at 0x0)
        Now 3 monitors running!
  ↓
...
  ↓
Seed 99: Creates monitor₉₉
         Now 100 monitors running! 😱
\end{textblock}

\end{record}

\subsubsection{Why Tests Still Passed}\label{cr:fix:pc-mismatch-fix:why-tests-still-passed}

\begin{record}

\begin{itemize}
\tightlist
\item
  Only the \textbf{current seed\textquotesingle s scoreboard} is checked
  for pass/fail
\item
  Old monitors log errors but their scoreboards are abandoned
\item
  Final scoreboard report only shows the active seed\textquotesingle s
  results
\end{itemize}

\begin{center}\rule{0.5\linewidth}{0.5pt}\end{center}

\end{record}

\subsection{The Fix}\label{cr:fix:pc-mismatch-fix:the-fix}

\subsubsection{Implementation}\label{cr:fix:pc-mismatch-fix:implementation}

  \end{lstlisting}
\item

  \begin{itemize}
  \tightlist
  \item

  \item
    Task accumulation causes performance degradation
  \item
    Stale state causes confusing error messages
  \end{itemize}
\end{enumerate}

\end{record}

\textbf{Problem}: EBREAK instruction caused CPU to halt, but halt cause
was reported incorrectly (0x0 instead of 0x8)

\textbf{Root Cause}:

\begin{itemize}
\tightlist
\item
  EBREAK transitions directly from EXECUTE→HALTED (skips WRITEBACK)
\item
  Original detection logic relied on \code{commit_valid}, which is only
  asserted in WRITEBACK
\item
  Result: \code{ebreak_halt} flag was never set
\end{itemize}

\textbf{Solution}:

\begin{itemize}
\tightlist
\item
  Exposed \code{ebreak} signal from decoder through core hierarchy
  (\code{debug_ebreak})
\item
  Implemented two-stage detection:

  \begin{enumerate}
  \def\labelenumi{\arabic{enumi}.}
  \tightlist
  \item
    Latch when \code{ebreak} is decoded (while CPU executing)
  \item
    Set \code{ebreak_halt} when CPU enters HALTED state
  \end{enumerate}
\item
  Ensures reliable detection of EBREAK-caused halts
\end{itemize}

\subsubsection{EBREAK Fix Implementation}\label{cr:app:archive-session-summary:ebreak-fix-implementation}

\begin{record}

\textbf{Signal Flow}:

\begin{lstlisting}[language={}]
rv32i_decode.sv:
  - Decodes EBREAK (0x00100073)
  - Sets ebreak = 1

rv32i_core.sv:
  - Exposes ebreak as debug_ebreak output
  - Connects to control FSM

rv32i_cpu_top.sv:
  - Two-stage detection:
    1. ebreak_caused_halt <= debug_ebreak (when executing)
    2. ebreak_halt <= ebreak_caused_halt (when halted)
  - Status register reports correct cause (0x8)
\end{lstlisting}

\textbf{State Transitions}:

\begin{textblock}
Normal instruction:
  FETCH → DECODE → EXECUTE → MEM_WAIT/WRITEBACK → FETCH

EBREAK instruction:
  FETCH → DECODE → EXECUTE → HALTED ✓
                           (skips WRITEBACK)
\end{textblock}

\end{record}
