\section{Verification Plan}\label{ds:ch:verification-plan}

Phase-Aligned Verification Strategy with AI/Human Responsibilities

Document status: Frozen Last updated: 2026-01-17

\subsection{Overview}\label{ds:ch:verification-plan:overview}

This document defines the verification strategy for all project phases,
detailing what to verify, how to verify it, and clear AI vs Human
responsibilities for verification tasks.

\subsubsection{Core Principles}\label{ds:ch:verification-plan:core-principles}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Python is the golden reference}: Reference models define
  correctness
\item
  \textbf{Scoreboards decide truth}: RTL behavior compared against
  reference models
\item
  \textbf{Randomized stress \textgreater{} Directed tests}: After smoke
  tests pass, emphasize random testing
\item
  \textbf{Verification precedes optimization}: Never optimize unverified
  RTL
\item
  \textbf{Specification-driven}: All verification derives from
  architectural specs
\end{enumerate}

\subsubsection{Verification Stack}\label{ds:ch:verification-plan:verification-stack}

\textbf{Tools}:

\begin{itemize}
\tightlist
\item
  \textbf{cocotb}: Signal-level drivers, monitors, clocking, resets
\item
  \textbf{pyuvm}: Structure, sequences, agents, scoreboards
\item
  \textbf{Python reference models}: Golden behavioral models (see
  REFERENCE\_\allowbreak{}MODEL\_\allowbreak{}SPEC.md)
\item
  \textbf{pytest}: Unit testing for reference models and testbench
  components
\end{itemize}

\textbf{Simulation}:

\begin{itemize}
\tightlist
\item
  \textbf{Verilator}: Fast, free, linting
\item
  \textbf{Questa/\allowbreak{}Xcelium}: Full-featured (if available)
\end{itemize}

\subsubsection{Scoreboard Implementation}\label{ds:ch:verification-plan:scoreboard-implementation}

\begin{lstlisting}[language=Python]
class CPUScoreboard(uvm_component):
    def __init__(self, name, parent):
        super().__init__(name, parent)
        self.model = RV32IModel()  # Reference model
        self.mismatches = 0

    def write_commit(self, txn):
        """Called when RTL commits an instruction"""
        # Step reference model with same instruction
        ref_result = self.model.step(txn.insn)

        # Compare PC
        if txn.pc != ref_result['pc']:
            self.log.error(f"PC mismatch: RTL={txn.pc:08x}, Model={ref_result['pc']:08x}")
            self.mismatches += 1

        # Compare register writes
        if ref_result['rd'] is not None:
            rtl_rd_val = self.get_rtl_register(ref_result['rd'])
            if rtl_rd_val != ref_result['rd_value']:
                self.log.error(f"Register x{ref_result['rd']} mismatch")
                self.mismatches += 1

        # Compare memory writes
        if ref_result['mem_write']:
            # Check memory write matches
            pass  # Implementation details

    def report_phase(self):
        if self.mismatches == 0:
            self.log.info("PASS: No mismatches detected")
        else:
            self.log.error(f"FAIL: {self.mismatches} mismatches")
\end{lstlisting}

\textbf{Human responsibility}: Design and implement scoreboard logic

\subsubsection{Assertions (SystemVerilog)}\label{ds:ch:verification-plan:assertions-systemverilog}

\textbf{Note}: SVA assertions in separate files, not in RTL

\textbf{Critical assertions}:

\begin{lstlisting}[language=Verilog]
// No double write-back in same cycle
assert property (@(posedge clk) disable iff (!rst_n)
    commit_valid |-> ##1 !commit_valid || (commit_pc != $past(commit_pc))
);

// x0 always zero
assert property (@(posedge clk) disable iff (!rst_n)
    regfile.regs[0] == 0
);

// AXI4-Lite protocol: valid must not deassert after assert until ready
assert property (@(posedge clk) disable iff (!rst_n)
    $rose(axi_arvalid) |-> axi_arvalid throughout axi_arready[->1]
);
\end{lstlisting}

\textbf{AI may}: Generate assertion templates

\subsubsection{Debugging Strategy}\label{ds:ch:verification-plan:debugging-strategy}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Check scoreboard logs}: Identify first mismatch
\item
  \textbf{Dump waveforms}: Examine cycle-by-cycle behavior
\item
  \textbf{Compare with reference model}: Step through both RTL and model
\item
  \textbf{Simplify test case}: Minimize failing sequence
\item
  \textbf{Add targeted assertions}: Catch bug earlier
\end{enumerate}

\textbf{AI may}: Help analyze logs, suggest debugging approaches

\textbf{Human must}: Root-cause failures, fix RTL or model
