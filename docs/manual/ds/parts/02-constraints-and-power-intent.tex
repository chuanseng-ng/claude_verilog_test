Document status: Phase 1 integration Target audience: Physical design
engineers, timing engineers, STA analysts Standard: Synopsys Design

\subsection{Overview}\label{ds:ch:sdc-timing-spec:overview}

This document specifies timing constraints for the RV32I+GPU SoC using
Synopsys Design Constraints (SDC) format. Timing constraints are
essential for static timing analysis (STA) and guide synthesis and place
\& route tools.

\subsubsection{Timing Verification Goals}\label{ds:ch:sdc-timing-spec:timing-verification-goals}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Setup timing}: Ensure data arrives before clock edge (with
  margin)
\item
  \textbf{Hold timing}: Ensure data remains stable after clock edge
\item
  \textbf{Clock quality}: Minimize skew and jitter
\item
  \textbf{Interface timing}: Meet AXI, APB protocol requirements
\end{enumerate}

\subsubsection{Timing Closure Strategy}\label{ds:ch:sdc-timing-spec:timing-closure-strategy}

\begin{itemize}
\tightlist
\item
  \textbf{Phase 1}: 100 MHz (10ns period), single clock, simple
  constraints
\item
  \textbf{Phase 2}: 200-500 MHz (5-2ns period), pipeline timing critical
\item
  \textbf{Phase 3}: Cache access timing, SRAM models
\item
  \textbf{Phase 4}: Multi-clock domains (CPU clock + GPU clock)
\item
  \textbf{Phase 5}: Full SoC with multiple asynchronous clocks
\end{itemize}

\subsection{SDC Concepts}\label{ds:ch:sdc-timing-spec:sdc-concepts}

\subsubsection{Clock Definition}\label{ds:ch:sdc-timing-spec:clock-definition}

A \textbf{clock} defines the periodic signal that synchronizes the
design.

\begin{lstlisting}[language=tcl]
create_clock -name clk -period 10.0 [get_ports clk]
\end{lstlisting}

\textbf{Parameters}:

\begin{itemize}
\tightlist
\item
  \code{-name}: Clock name (for reference)
\item
  \code{-period}: Clock period in ns
\item
  \texttt{[get\_\allowbreak{}ports\ clk]}: Clock source (port or pin)
\end{itemize}

\subsubsection{Clock Uncertainty}\label{ds:ch:sdc-timing-spec:clock-uncertainty}

\textbf{Clock uncertainty} accounts for jitter and skew in the clock
network.

\begin{lstlisting}[language=tcl]
set_clock_uncertainty 0.5 [get_clocks clk]
\end{lstlisting}

\textbf{Typical values}:

\begin{itemize}
\tightlist
\item
  \textbf{Pre-CTS}: 5-10\% of period (conservative estimate)
\item
  \textbf{Post-CTS}: 1-3\% of period (after clock tree is built)
\end{itemize}

\subsubsection{Clock Latency}\label{ds:ch:sdc-timing-spec:clock-latency}

\textbf{Clock latency} is the delay from clock source to register clock
pins.

\begin{lstlisting}[language=tcl]
# Source latency (external delay before chip)
set_clock_latency -source 1.0 [get_clocks clk]

# Network latency (internal clock tree delay)
set_clock_latency 0.5 [get_clocks clk]
\end{lstlisting}

\subsubsection{\texorpdfstring{Input/\allowbreak{}Output
Delays}{Input/Output Delays}}\label{ds:ch:sdc-timing-spec:inputoutput-delays}

\textbf{Input delay}: Time from external clock edge to data arrival at
input port \textbf{Output delay}: Time from output port to external
register clock pin

\begin{lstlisting}[language=tcl]
# Input delay (external device has tSU = 2ns)
set_input_delay -clock clk -max 2.0 [get_ports data_in]
set_input_delay -clock clk -min 0.5 [get_ports data_in]

# Output delay (external device has tH = 1ns)
set_output_delay -clock clk -max 2.0 [get_ports data_out]
set_output_delay -clock clk -min 0.5 [get_ports data_out]
\end{lstlisting}

\subsubsection{False Paths}\label{ds:ch:sdc-timing-spec:false-paths}

\textbf{False paths} are signal paths that do not require timing
analysis.

\textbf{Examples}:

\begin{itemize}
\tightlist
\item
  Asynchronous reset paths
\item
  Paths between unrelated clock domains
\item
  Test/\allowbreak{}debug paths
\end{itemize}

\begin{lstlisting}[language=tcl]
# Async reset is a false path
set_false_path -from [get_ports rst_n]

# Debug interface doesn't need to meet AXI timing
set_false_path -from [get_ports apb_*] -to [get_ports axi_*]
\end{lstlisting}

\subsubsection{Multi-Cycle Paths}\label{ds:ch:sdc-timing-spec:multi-cycle-paths}

\textbf{Multi-cycle paths} are paths that have more than one clock cycle
to propagate.

\textbf{Example}: A slow control path that takes 2 cycles

\begin{lstlisting}[language=tcl]
# This path has 2 cycles instead of 1
set_multicycle_path 2 -from [get_pins ctrl_reg*] -to [get_pins next_state*]
set_multicycle_path 1 -hold -from [get_pins ctrl_reg*] -to [get_pins next_state*]
\end{lstlisting}

\textbf{Note}: Always specify both setup (default) and hold multicycle
paths.

\subsubsection{Clock Groups}\label{ds:ch:sdc-timing-spec:clock-groups}

\textbf{Clock groups} define relationships between clocks (synchronous,
asynchronous, exclusive).

\begin{lstlisting}[language=tcl]
# Define asynchronous clock groups (no timing checks between them)
set_clock_groups -asynchronous \
    -group [get_clocks clk_cpu] \
    -group [get_clocks clk_gpu]
\end{lstlisting}

phase1\_\allowbreak{}cpu.sdc}{SDC File: phase1\_cpu.sdc}}\label{ds:ch:sdc-timing-spec:sdc-file-phase1-cpusdc}

\begin{lstlisting}[language=tcl]
#=============================================
# Phase 1: Single-cycle RV32I CPU
# Target: 100 MHz (10ns period)
#=============================================

#---------------------------------------------
# Clock Definition
#---------------------------------------------

# Create primary clock
create_clock -name clk -period 10.0 [get_ports clk]

# Clock uncertainty (pre-CTS)
set_clock_uncertainty 0.5 [get_clocks clk]

# Clock latency (source: external PLL)
set_clock_latency -source 1.0 [get_clocks clk]

# Clock transition time (slew)
set_clock_transition 0.1 [get_clocks clk]

#---------------------------------------------
# Input Constraints
#---------------------------------------------

# AXI4-Lite input delays (based on protocol timing)
# Assume external memory controller with 2ns setup, 0.5ns hold
set_input_delay -clock clk -max 2.0 [get_ports axi_*ready]
set_input_delay -clock clk -max 2.0 [get_ports axi_rdata]
set_input_delay -clock clk -max 2.0 [get_ports axi_rresp]
set_input_delay -clock clk -max 2.0 [get_ports axi_rvalid]
set_input_delay -clock clk -max 2.0 [get_ports axi_bvalid]
set_input_delay -clock clk -max 2.0 [get_ports axi_bresp]

set_input_delay -clock clk -min 0.5 [get_ports axi_*ready]
set_input_delay -clock clk -min 0.5 [get_ports axi_rdata]
set_input_delay -clock clk -min 0.5 [get_ports axi_rresp]
set_input_delay -clock clk -min 0.5 [get_ports axi_rvalid]
set_input_delay -clock clk -min 0.5 [get_ports axi_bvalid]
set_input_delay -clock clk -min 0.5 [get_ports axi_bresp]

# APB3 input delays (debug interface, less critical)
set_input_delay -clock clk -max 3.0 [get_ports apb_paddr]
set_input_delay -clock clk -max 3.0 [get_ports apb_psel]
set_input_delay -clock clk -max 3.0 [get_ports apb_penable]
set_input_delay -clock clk -max 3.0 [get_ports apb_pwrite]
set_input_delay -clock clk -max 3.0 [get_ports apb_pwdata]

set_input_delay -clock clk -min 1.0 [get_ports apb_*]

#---------------------------------------------
# Output Constraints
#---------------------------------------------

# AXI4-Lite output delays
set_output_delay -clock clk -max 2.0 [get_ports axi_awaddr]
set_output_delay -clock clk -max 2.0 [get_ports axi_awvalid]
set_output_delay -clock clk -max 2.0 [get_ports axi_wdata]
set_output_delay -clock clk -max 2.0 [get_ports axi_wstrb]
set_output_delay -clock clk -max 2.0 [get_ports axi_wvalid]
set_output_delay -clock clk -max 2.0 [get_ports axi_araddr]
set_output_delay -clock clk -max 2.0 [get_ports axi_arvalid]
set_output_delay -clock clk -max 2.0 [get_ports axi_rready]
set_output_delay -clock clk -max 2.0 [get_ports axi_bready]

set_output_delay -clock clk -min 0.5 [all_outputs]

# APB3 output delays
set_output_delay -clock clk -max 3.0 [get_ports apb_prdata]
set_output_delay -clock clk -max 3.0 [get_ports apb_pready]
set_output_delay -clock clk -max 3.0 [get_ports apb_pslverr]

# Commit interface (verification only, relaxed)
set_output_delay -clock clk -max 5.0 [get_ports commit_*]
set_output_delay -clock clk -max 5.0 [get_ports trap_*]

#---------------------------------------------
# False Paths
#---------------------------------------------

# Asynchronous reset
set_false_path -from [get_ports rst_n]

# Debug paths don't affect AXI timing
set_false_path -from [get_ports apb_*] -to [get_ports axi_*]
set_false_path -from [get_ports axi_*] -to [get_ports apb_*]

# Commit interface is for verification only
set_false_path -to [get_ports commit_*]
set_false_path -to [get_ports trap_*]

#---------------------------------------------
# Load and Drive Constraints
#---------------------------------------------

# Output load (assume 50fF capacitance)
set_load 0.05 [all_outputs]

# Input drive (assume BUF_X1 driving inputs)
set_driving_cell -lib_cell BUF_X1 -library sky130_fd_sc_hd [all_inputs]

#---------------------------------------------
# Area Constraint (for synthesis)
#---------------------------------------------

# No specific area constraint (optimize for timing)
set_max_area 0

#---------------------------------------------
# Environment
#---------------------------------------------

# Operating conditions
set_operating_conditions -max typical -min typical

# Wire load model (for pre-layout synthesis)
# set_wire_load_model -name "conservative" -library sky130_fd_sc_hd

#---------------------------------------------
# Design Rule Constraints
#---------------------------------------------

# Max transition time (slew rate)
set_max_transition 0.5 [current_design]

# Max capacitance
set_max_capacitance 0.1 [all_outputs]

# Max fanout
set_max_fanout 16 [current_design]
\end{lstlisting}

# Phase 2: Pipelined CPU
# Target: 200 MHz (5ns period)
#=============================================

# Higher frequency clock
create_clock -name clk -period 5.0 [get_ports clk]

# Tighter uncertainty
set_clock_uncertainty 0.25 [get_clocks clk]

# Control FSM may take 2 cycles to update
set_multicycle_path 2 -setup -from [get_pins i_rv32i_control/state_reg*] \
                              -to [get_pins i_rv32i_control/next_state*]

                              -to [get_pins i_rv32i_control/next_state*]

# These should meet single-cycle timing
set_max_delay 5.0 -from [get_pins */fetch_stage/*] -to [get_pins */decode_stage/*]
set_max_delay 5.0 -from [get_pins */decode_stage/*] -to [get_pins */execute_stage/*]
set_max_delay 5.0 -from [get_pins */execute_stage/*] -to [get_pins */memory_stage/*]
set_max_delay 5.0 -from [get_pins */memory_stage/*] -to [get_pins */writeback_stage/*]

# Forwarding paths (critical for hazard handling)
set_max_delay 4.0 -from [get_pins */execute_stage/alu_result*] \
                   -to [get_pins */execute_stage/forward_mux*]
\end{lstlisting}

Document status: Phase 1 integration Target audience: Physical design

\subsection{Overview}\label{ds:ch:upf-power-spec:overview}

This document specifies the power intent for the RV32I+GPU SoC using
Unified Power Format (UPF). Power management is introduced incrementally
across phases to balance complexity and power savings.

\subsubsection{Power Management Goals}\label{ds:ch:upf-power-spec:power-management-goals}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Reduce dynamic power}: Clock gating, power gating idle blocks
\item
  \textbf{Reduce leakage power}: Power down unused domains
\item
  \textbf{Preserve functionality}: Isolation and retention strategies
\item
  \textbf{Enable flexibility}: Multiple power modes for different
  workloads
\end{enumerate}

\subsubsection{Incremental Power Strategy}\label{ds:ch:upf-power-spec:incremental-power-strategy}

\begin{itemize}
\tightlist
\item
  \textbf{Phase 1}: Single domain (prepare infrastructure, no power
  gating yet)
\item
  \textbf{Phase 2}: Core + debug domains (enable basic power gating)
\item
  \textbf{Phase 3}: Core + I-cache + D-cache domains (cache power
  management)
\item
  \textbf{Phase 4}: Add GPU domain with DVFS
  (voltage/\allowbreak{}frequency scaling)
\item
  \textbf{Phase 5}: Full system with 5+ domains (always-on, peripherals,
  etc.)
\end{itemize}

\subsection{UPF Concepts}\label{ds:ch:upf-power-spec:upf-concepts}

\subsubsection{Power Domains}\label{ds:ch:upf-power-spec:power-domains}

A \textbf{power domain} is a collection of design elements that share
the same power supply and can be powered on/off together.

\textbf{Example}: CPU core, debug interface, caches, GPU, peripherals

\subsubsection{Supply Networks}\label{ds:ch:upf-power-spec:supply-networks}

A \textbf{supply network} defines the power (VDD) and ground (VSS) nets
for a domain.

\textbf{Example}: \code{VDD_CORE}, \code{VDD_GPU}, \code{VDD_AON}
(always-on), \code{VSS}

\subsubsection{Isolation Cells}\label{ds:ch:upf-power-spec:isolation-cells}

\textbf{Isolation cells} prevent X (unknown) values from propagating
from a powered-down domain to a powered-on domain.

\textbf{Location}: At domain boundaries (outputs of powered-down domain)
\textbf{Clamp value}: 0 or 1 (designer specifies safe value)
\textbf{Control}: Enabled when source domain is powered down

\subsubsection{Level Shifters}\label{ds:ch:upf-power-spec:level-shifters}

\textbf{Level shifters} convert signals between domains operating at
different voltages.

\textbf{Example}: Core at 0.9V, peripherals at 1.8V

\textbf{Direction}: Low-to-high or high-to-low

\subsubsection{Retention Registers}\label{ds:ch:upf-power-spec:retention-registers}

\textbf{Retention registers} preserve their state during power-down
using a separate always-on supply.

\textbf{Use case}: Save critical state (PC, control registers) during
sleep mode

\textbf{Restore}: State restored automatically when domain powers back
on

\subsubsection{Power State Tables (PST)}\label{ds:ch:upf-power-spec:power-state-tables-pst}

A \textbf{power state table} defines the legal combinations of power
states across domains.

\textbf{States}:

\begin{itemize}
\tightlist
\item
  \code{ON}: Domain fully powered
\item
  \code{OFF}: Domain powered down
\item
  \code{RETENTION}: Domain powered down but state preserved
\end{itemize}

\subsubsection{Power Mode Transitions (Phase 2)}\label{ds:ch:upf-power-spec:power-mode-transitions-phase-2}

\textbf{ACTIVE → IDLE}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  CPU idle (WFI instruction or no work)
\item
  Clock gating enabled automatically
\item
  No state change required
\end{enumerate}

\textbf{IDLE → SLEEP}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  Software writes to power control register
\item
  Assert \code{save_core} signal (save state to retention cells)
\item
  Assert \code{iso_core_en} (enable isolation)
\item
  Assert \code{ret_core_en} (enable retention supply)
\item
  Gate \code{VDD_CORE} supply
\end{enumerate}

\textbf{SLEEP → ACTIVE}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  Interrupt or wake event
\item
  Ungate \code{VDD_CORE} supply
\item
  Wait for supply to stabilize
\item
  Assert \code{restore_core} (restore state from retention cells)
\item
  Deassert \code{ret_core_en}
\item
  Deassert \code{iso_core_en}
\item
  Resume execution
\end{enumerate}

checker fails if a top-level instance is added to or removed from
\code{soc_top} without updating the roster.

\subsubsection{What is modelled, and what is deliberately not}\label{ds:ch:upf-power-spec:what-is-modelled-and-what-is-deliberately-not}

\begin{itemize}
\tightlist
\item
  \textbf{Power switches: none.} \code{pmu.sv} emits no switch control
  (\code{cpu_pg_ctrl} / \code{gpu_pg_ctrl} do not exist) and no
  \code{soc_top} net could drive one. The old \code{create_power_switch}
  commands controlled ports that were never in the RTL and used options
  no UPF parser accepts. \code{VDD_CPU} / \code{VDD_GPU} are therefore
  modelled as externally driven supply ports with ON / OFF port states,
  so each supply net has a source. The UPF header carries the shape of
  the switch to add once RTL control exists.
\item
  \textbf{Isolation: exactly the signals \code{soc_top} clamps}, not
  every output. \code{ISO_CPU} (6 signals, control
  \code{pmu_cpu_iso_en_sync}) and \code{ISO_GPU} (8 signals, control
  \code{pmu_gpu_iso_en}), clamp value 0, sense high,
  \texttt{-\allowbreak{}location\ parent}, supply \code{SS_AON}. The set
  is checked for equality with the RTL in both SRAM arms. AXI
  address/\allowbreak{}data/\allowbreak{}control buses stay unclamped
  because each is qualified by a clamped valid and is
  don\textquotesingle t-care once valid is 0. Every other output of
  \code{u_cpu} / \code{u_gpu} is classified in the UPF (\code{payload},
  \code{slave_response}, \code{observability}) so a new output cannot
  escape review.
\item
  \textbf{PMU outputs are flop Q pins (bead \bead{ktwt}).}
  \code{pmu_*_clk_en} / \code{iso_en} / \code{rst_n} / \code{ret_*} are
  each registered inside \code{pmu.sv} from the decode of the
  \emph{next} FSM state, so they appear in exactly the cycle the earlier
  combinational decode showed them (the sequence above and
  \code{test_pmu}\textquotesingle s per-cycle expectations are
  unchanged: zero added latency). They feed \code{cdc_sync} D pins, the
  asynchronous clear of \code{u_cpu_pmu_rst_sync} and the GPU async
  reset / clamp select, where a decode glitch on the 4-bit binary state
  (possible on 4 of the 9 transitions, including a runt reset on
  \code{DOM_PU_RESTORE} -\textgreater{} \code{DOM_PU_ISO}) would be a
  real event. Zero-delay simulation cannot show a glitch, so
  \src{tools/verif/check_registered_outputs.py} proves it on the netlist
  (\src{tb/tests/test_pmu_registered_outputs.py}).
\item
  \textbf{Retention: none.} The old header claim that state is "rebuilt
  from SRAM on wakeup" was never true. \code{pmu.sv} emits
  \code{ret_save} / \code{ret_restore} in order but \code{soc_top}
  leaves them unconnected. See \src{docs/POWER_DOMAIN_EVALUATION.md} (\cref{tm:pd:power-domain-evaluation})
  section 3 for what survives a PMU off/on cycle.
\item
  \textbf{Not expressed} because the RTL or library offers nothing to
  bind: macro PG-pin connections (done by \code{pnr/*/pdn*.tcl}), level
  shifters (single voltage), isolation and retention cell mapping
  (neither ASAP7 nor Sky130 ships an isolation cell).
\end{itemize}

\subsubsection{Known gaps and follow-ups}\label{ds:ch:upf-power-spec:known-gaps-and-follow-ups}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Isolation-versus-reset ordering (was: no handshake; FIXED,
  bead \bead{3ogd}).} The PMU walks the UPF order (down: save, isolate,
  gate, reset; up: reset release, ungate, restore, de-isolate). The CPU
  clamp control passes a 2-flop synchroniser into \code{cpu_core_clk}
  while the reset reaches \code{u_cpu} through a \code{cdc_reset_sync}
  with asynchronous assertion, so with the actions merely spaced one
  \code{core_clk} cycle apart "isolate before reset asserts" held only
  while two \code{cpu_core_clk} edges were shorter than two
  \code{core_clk} cycles (it failed at a CPU clock 3x slower).
  \code{soc_top.u_cpu_iso_ack_sync} now synchronises the CPU-domain
  clamp control back into \code{core_clk} (\code{pmu_cpu_iso_ack}) and
  the CPU \code{DOM_PD_ISO} state waits for it before the clock gate and
  the reset; \code{DOM_PU_ISO} waits for it to fall (return-to-zero,
  which also stops a stale ack from satisfying the next power-down).
  Power-up needs no ack for the iso-release-vs-reset-release order: the
  clamp\textquotesingle s consumers are clocked by \code{cpu_gated_clk},
  which cannot tick before the reset release is observed
  (\code{cpu_gated_clk_en} includes \code{pmu_cpu_rst_n_cpu_sync}). The
  GPU needs none (same clock, the clamp control is the PMU flop). The
  step order of this UPF is unchanged. A dead CPU clock (CPU PLL
  unlocked) never acks: \code{ISO_ACK_TIMEOUT} (255 \code{core_clk}
  cycles) lets the PMU proceed and sets \code{STATUS[3]}.
\item
  \textbf{Slave-side responses are not isolated} (\code{u_gpu} AXI-Lite
  \code{rvalid}, \code{awready}, ... and \code{u_cpu} debug APB
  \code{pready} ...). With a really removed supply they would drive X
  into the always-on ring.
\item
  \textbf{Scan does not force the clamps off} (table above).
\item
  \code{m_axil_if_rready} was the one GPU-to-fabric handshake the RTL
  did not clamp; clamped as part of \code{h3lm}.
\end{enumerate}

\subsubsection{Checking}\label{ds:ch:upf-power-spec:checking}

\texttt{python3\ tools/\allowbreak{}verif/\allowbreak{}check\_\allowbreak{}upf.\allowbreak{}py}
(stdlib only, CI gate in \code{qa-checks.yml}) verifies that every
\code{-elements} path and control net resolves in the elaborated
\code{soc_top}, that the UPF clamp set equals the RTL clamp set, that
domain membership is explicit and disjoint, and that state names,
encodings and per-mode rail states equal \code{pmu.sv}. It reads
\src{pnr/constraints/phase5_soc_rtl_facts.json}, which
\texttt{check\_\allowbreak{}upf.\allowbreak{}py\ gen-\allowbreak{}facts}
generates from \src{rtl/filelist/soc_top.f} with pyslang for both SRAM
arms (ASAP7 and Sky130);
\texttt{gen-\allowbreak{}facts\ -\allowbreak{}-\allowbreak{}check} is
the drift gate. A passing run proves \textbf{references and set
equalities, not tool acceptance and not electrical sufficiency of a
clamp.}

\subsubsection{Tool compatibility}\label{ds:ch:upf-power-spec:tool-compatibility}

Measured 2026-10-17 on OpenROAD 26Q2:
\texttt{read\_\allowbreak{}upf\ -\allowbreak{}file} accepts only a small
subset (it defines \code{create_power_domain}, \code{set_isolation},
\code{create_power_switch}, \code{define_power_switch_cell} and
\code{map_power_switch}; there is no \code{upf_version},
\code{create_supply_*}, \code{add_power_state}, ...). It fails on the
first command of both the old file (\code{create_supply_port}) and the
new one (\code{upf_version}). The file targets full UPF 3.0, so this is
a tool-subset limitation and no gate depends on it. No UPF-aware
simulator or linter is available in the open flow.

\subsection{Power-Aware Verification}\label{ds:ch:upf-power-spec:power-aware-verification}

\subsubsection{UPF Simulation Flow}\label{ds:ch:upf-power-spec:upf-simulation-flow}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  \textbf{Compile with UPF}: Add UPF file to simulation

  \begin{lstlisting}[language=bash]
  verilator --upf phase2_pipelined_cpu.upf ...
  \end{lstlisting}
\item
  \textbf{Apply power sequences}: Use testbench to control power
  supplies

  \begin{textblock}
  # Example: cocotb power sequence
  await power_ctrl.power_down_core()  # ACTIVE → SLEEP
  await Timer(100, units='us')
  await power_ctrl.power_up_core()    # SLEEP → ACTIVE
  \end{textblock}
\item
  \textbf{Check isolation}: Verify no X propagation during power-down

  \begin{lstlisting}[language=Python]
  assert dut.axi_arvalid.value != "x", "Isolation failed!"
  \end{lstlisting}
\item
  \textbf{Check retention}: Verify state preserved after power cycle

  \begin{lstlisting}[language=Python]
  pc_before = dut.i_rv32i_core.pc_reg.value
  await power_cycle()
  pc_after = dut.i_rv32i_core.pc_reg.value
  assert pc_before == pc_after, "Retention failed!"
  \end{lstlisting}
\end{enumerate}

\subsubsection{UPF Testbench Structure}\label{ds:ch:upf-power-spec:upf-testbench-structure}

\begin{lstlisting}[language=Python]
# cocotb testbench with UPF support

import cocotb
from cocotb.triggers import Timer, RisingEdge

@cocotb.test()
async def test_power_down_core(dut):
    """Test core power-down and isolation"""

    # Initialize
    await reset_sequence(dut)

    # Run some instructions
    await run_instructions(dut, count=100)

    # Save PC value
    pc_before = dut.i_rv32i_core.pc_reg.value

    # Power down core
    dut.VDD_CORE.value = 0
    await Timer(1, units='us')

    # Check isolation (outputs should be clamped)
    assert dut.commit_valid.value == 0, "Isolation failed"

    # Power up core
    dut.VDD_CORE.value = 1
    await Timer(1, units='us')

    # Check retention (PC should be preserved)
    pc_after = dut.i_rv32i_core.pc_reg.value
    assert pc_before == pc_after, "Retention failed"

    # Resume execution
    await run_instructions(dut, count=100)
\end{lstlisting}

\subsection{UPF Best Practices}\label{ds:ch:upf-power-spec:upf-best-practices}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Start simple}: Phase 1 has single domain, add complexity
  incrementally
\item
  \textbf{Isolate everything}: Always add isolation at domain boundaries
\item
  \textbf{Retain critical state}: PC, control registers, cache tags
\item
  \textbf{Test extensively}: Power-aware simulation catches 90\% of UPF
  bugs
\item
  \textbf{Document power modes}: Clear PST with mode descriptions
\item
  \textbf{Use standard cells}: Isolation, retention, level shifter cells
  from PDK
\end{enumerate}
